# ============================================================================
# font_defs_CN.txt - 字体定义解析器DSL (中文注释版)
# ============================================================================
# 此文件定义了游戏字体配置文件的解析规则，用于ParserGenerator生成解析器代码。
# 生成的解析器代码保存在 font_defs_generated.hpp 中。
#
# DSL格式说明：
#   第一行：根结构名称
#   后续行：字段定义，格式为
#       字段名  值类型  数据类型  处理方式
#
# 值类型(value)：该字段是一个值属性（key = value 形式）
# 组类型(group)：该字段是一个分组（key { ... } 形式，内含子字段）
# 无类型(none)：该字段无值，仅作为标记
#
# 处理方式：
#   member_fn：调用成员函数处理
#   member：直接赋值给成员变量
#   discard：忽略该字段，不处理
#
# 注意：此文件为注释版副本，原始文件为 font_defs.txt
#       请勿将本文件作为ParserGenerator的输入使用
# ============================================================================

# 字体主体定义：单个字体的配置信息
font_body
	# 字体ID（值类型，无符号整数，直接赋值给成员变量）
	id				value		uint		member
	# 是否启用小型大写字母（值类型，布尔，直接赋值给成员变量）
	smallcaps		value		bool		member
	# 字体文件路径（值类型，文本，成员函数处理）
	path			value		text		member_fn
	# 后备字体名称（值类型，文本，成员函数处理）
	fallback		value		text		member_fn

# 字体文件定义：字体配置文件的根结构
font_file
	# 字体配置组（组类型，font_body分组，成员函数处理）
	font			group		font_body	member_fn
	# 是否使用黑色地图字体（值类型，布尔，直接赋值给成员变量）
	blackmapfont	value		bool		member
